cmake_minimum_required(VERSION 3.18)
project(vkfft_direct_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VKFFT_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../VkFFT/vkFFT" CACHE PATH "Unmodified upstream include directory")
option(DIRECT_CPU_ONLY "Build only candidate validation tests" OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(test_direct_candidate test_direct_candidate.cpp)
target_include_directories(test_direct_candidate PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  # Avoid picking up incompatible GCC or winpthread DLLs from an unrelated PATH.
  target_link_options(test_direct_candidate PRIVATE -static)
endif()
add_test(NAME direct_candidate_validation COMMAND test_direct_candidate)
add_test(NAME direct_generation_anchors COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_generation.py")
if(NOT DIRECT_CPU_ONLY)
  find_package(CUDAToolkit REQUIRED)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_direct_headers.py"
    --source "${VKFFT_INCLUDE_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated/include"
    RESULT_VARIABLE PREPARE_RESULT)
  if(NOT PREPARE_RESULT EQUAL 0)
    message(FATAL_ERROR "Direct candidate header generation failed")
  endif()
  add_executable(vkfft_direct_candidate "${CMAKE_CURRENT_BINARY_DIR}/generated/direct_benchmark.cpp")
  target_include_directories(vkfft_direct_candidate PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/include" "${CMAKE_CURRENT_SOURCE_DIR}/../instrumentation")
  target_compile_definitions(vkfft_direct_candidate PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_direct_candidate PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_direct_candidate PRIVATE -O3)
  endif()
endif()
