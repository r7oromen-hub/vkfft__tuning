cmake_minimum_required(VERSION 3.18)
project(vkfft_precode_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PRECODE_GENERATED_ROOT "" CACHE PATH "Directory containing generated/direct_benchmark.cpp and generated/include")
option(PRECODE_CPU_ONLY "Build the synthetic trace snapshot test without CUDA" OFF)
if(NOT EXISTS "${PRECODE_GENERATED_ROOT}/direct_benchmark.cpp" OR NOT EXISTS "${PRECODE_GENERATED_ROOT}/include/trace.hpp")
  message(FATAL_ERROR "Generate the probe first and set PRECODE_GENERATED_ROOT to its parent directory")
endif()
if(PRECODE_CPU_ONLY)
  enable_testing()
  add_executable(test_probe_trace test_probe_trace.cpp)
  target_include_directories(test_probe_trace PRIVATE "${PRECODE_GENERATED_ROOT}/include")
  if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(test_probe_trace PRIVATE -static)
  endif()
  add_test(NAME precode_trace_snapshots COMMAND test_probe_trace)
  return()
endif()
find_package(CUDAToolkit REQUIRED)
add_executable(vkfft_precode_probe "${PRECODE_GENERATED_ROOT}/direct_benchmark.cpp")
target_include_directories(vkfft_precode_probe PRIVATE "${PRECODE_GENERATED_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../../instrumentation")
target_compile_definitions(vkfft_precode_probe PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
target_link_libraries(vkfft_precode_probe PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
if(NOT MSVC)
  target_compile_options(vkfft_precode_probe PRIVATE -O3)
endif()
