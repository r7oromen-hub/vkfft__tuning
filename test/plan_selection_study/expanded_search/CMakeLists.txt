cmake_minimum_required(VERSION 3.18)
project(vkfft_expanded_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(EXPANDED_CPU_ONLY "Build only CPU tests" OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(test_candidate test_candidate.cpp)
target_include_directories(test_candidate PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(test_candidate PRIVATE /UNDEBUG)
else()
  target_compile_options(test_candidate PRIVATE -UNDEBUG)
endif()
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(test_candidate PRIVATE -static)
endif()
add_test(NAME expanded_candidate_cpu COMMAND test_candidate)
add_test(NAME expanded_generation_cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_generation.py")
add_test(NAME expanded_enumeration_cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_enumeration.py")
if(NOT EXPANDED_CPU_ONLY)
  set(EXPANDED_GENERATED_ROOT "" CACHE PATH "Parent of generated include and expanded_benchmark.cpp")
  if(NOT EXISTS "${EXPANDED_GENERATED_ROOT}/include/candidate.hpp" OR NOT EXISTS "${EXPANDED_GENERATED_ROOT}/expanded_benchmark.cpp")
    message(FATAL_ERROR "Run prepare_headers.py into an empty output and set EXPANDED_GENERATED_ROOT to its parent")
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(vkfft_expanded_candidate "${EXPANDED_GENERATED_ROOT}/expanded_benchmark.cpp")
  target_include_directories(vkfft_expanded_candidate PRIVATE "${EXPANDED_GENERATED_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../instrumentation")
  target_compile_definitions(vkfft_expanded_candidate PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_expanded_candidate PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_expanded_candidate PRIVATE -O3)
  endif()
endif()
