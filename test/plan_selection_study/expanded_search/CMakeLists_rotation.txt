# Copy as CMakeLists.txt into an isolated build source directory, or use the
# generator's emitted copy. The original include root must be untouched.
cmake_minimum_required(VERSION 3.18)
project(vkfft_rotation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROTATION_GENERATED_ROOT "" CACHE PATH "Directory containing rotating_benchmark.cpp")
set(EXPANDED_GENERATED_ROOT "" CACHE PATH "Parent of frozen expanded generated include")
set(VKFFT_INCLUDE_ROOT "" CACHE PATH "Unmodified original VkFFT include root")
set(INSTRUMENTATION_ROOT "" CACHE PATH "instrumentation source directory")
if(NOT EXISTS "${ROTATION_GENERATED_ROOT}/rotating_benchmark.cpp" OR NOT EXISTS "${EXPANDED_GENERATED_ROOT}/include/candidate.hpp" OR NOT EXISTS "${VKFFT_INCLUDE_ROOT}/vkFFT.h")
  message(FATAL_ERROR "Generated rotation source, frozen expanded includes and original headers are required")
endif()
find_package(CUDAToolkit REQUIRED)
foreach(target IN ITEMS vkfft_rotation_expanded vkfft_rotation_original)
  add_executable(${target} "${ROTATION_GENERATED_ROOT}/rotating_benchmark.cpp")
  if(target STREQUAL "vkfft_rotation_original")
    target_include_directories(${target} PRIVATE "${VKFFT_INCLUDE_ROOT}")
    target_compile_definitions(${target} PRIVATE VKFFT_STUDY_UNINSTRUMENTED=1)
  endif()
  target_include_directories(${target} PRIVATE "${ROTATION_GENERATED_ROOT}" "${EXPANDED_GENERATED_ROOT}/include" "${INSTRUMENTATION_ROOT}")
  target_compile_definitions(${target} PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(${target} PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -O3)
  endif()
endforeach()
