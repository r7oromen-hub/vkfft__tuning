cmake_minimum_required(VERSION 3.18)
project(vkfft_frontier_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FRONTIER_CPU_ONLY "Build only CPU tests" OFF)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(test_frontier_candidate test_candidate.cpp)
target_include_directories(test_frontier_candidate PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(test_frontier_candidate PRIVATE /UNDEBUG)
else()
  target_compile_options(test_frontier_candidate PRIVATE -UNDEBUG)
endif()
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(test_frontier_candidate PRIVATE -static)
endif()
add_test(NAME frontier_candidate_cpu COMMAND test_frontier_candidate)
add_test(NAME frontier_python_cpu COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${CMAKE_CURRENT_SOURCE_DIR}" -p "test_*.py")
if(NOT FRONTIER_CPU_ONLY)
  set(FRONTIER_GENERATED_ROOT "" CACHE PATH "Parent of generated include and expanded_benchmark.cpp")
  get_filename_component(FRONTIER_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
  set(FRONTIER_ORIGINAL_INCLUDE_ROOT "${FRONTIER_REPOSITORY_ROOT}/VkFFT/vkFFT" CACHE PATH "Untouched VkFFT include root")
  if(NOT EXISTS "${FRONTIER_GENERATED_ROOT}/include/candidate.hpp" OR NOT EXISTS "${FRONTIER_GENERATED_ROOT}/expanded_benchmark.cpp")
    message(FATAL_ERROR "Run frontier prepare_headers.py and set FRONTIER_GENERATED_ROOT")
  endif()
  if(NOT EXISTS "${FRONTIER_ORIGINAL_INCLUDE_ROOT}/vkFFT.h")
    message(FATAL_ERROR "Original VkFFT include root not found")
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_executable(vkfft_frontier_v2 "${FRONTIER_GENERATED_ROOT}/expanded_benchmark.cpp")
  target_include_directories(vkfft_frontier_v2 PRIVATE "${FRONTIER_GENERATED_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../instrumentation")
  target_compile_definitions(vkfft_frontier_v2 PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_frontier_v2 PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_frontier_v2 PRIVATE -O3)
  endif()
  # Same generated benchmark source; only VkFFT headers and instrumentation
  # differ. Generated include supplies trace/source metadata and helpers.
  add_executable(vkfft_frontier_original "${FRONTIER_GENERATED_ROOT}/expanded_benchmark.cpp")
  target_include_directories(vkfft_frontier_original PRIVATE "${FRONTIER_ORIGINAL_INCLUDE_ROOT}" "${FRONTIER_GENERATED_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/../instrumentation")
  target_compile_definitions(vkfft_frontier_original PRIVATE VKFFT_BACKEND=1 VKFFT_STUDY_UNINSTRUMENTED=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_frontier_original PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_frontier_original PRIVATE -O3)
  endif()
endif()
