cmake_minimum_required(VERSION 3.18)
project(vkfft_instrumented_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VKFFT_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../VkFFT/vkFFT" CACHE PATH "Unmodified upstream include directory")
option(STUDY_CPU_ONLY "Only build CPU reference and trace tests" OFF)
enable_testing()
add_executable(test_reference test_reference.cpp)
add_test(NAME cpu_reference COMMAND test_reference)
add_executable(test_trace test_trace.cpp)
add_test(NAME trace_and_source_hash COMMAND test_trace)
if(NOT STUDY_CPU_ONLY)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(CUDAToolkit REQUIRED)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_headers.py"
    --source "${VKFFT_INCLUDE_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/generated/include"
    RESULT_VARIABLE PREPARE_RESULT)
  if(NOT PREPARE_RESULT EQUAL 0)
    message(FATAL_ERROR "Strict instrumented header generation failed")
  endif()
  add_executable(vkfft_instrumented_benchmark vkfft_instrumented_benchmark.cpp)
  target_include_directories(vkfft_instrumented_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/include" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(vkfft_instrumented_benchmark PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_instrumented_benchmark PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_instrumented_benchmark PRIVATE -O3)
  endif()
  add_executable(vkfft_uninstrumented_control vkfft_instrumented_benchmark.cpp)
  target_include_directories(vkfft_uninstrumented_control PRIVATE "${VKFFT_INCLUDE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated/include" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(vkfft_uninstrumented_control PRIVATE VKFFT_BACKEND=1 VKFFT_STUDY_UNINSTRUMENTED=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_uninstrumented_control PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_uninstrumented_control PRIVATE -O3)
  endif()
endif()
