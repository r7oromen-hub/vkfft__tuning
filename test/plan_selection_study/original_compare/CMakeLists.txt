cmake_minimum_required(VERSION 3.18)
project(vkfft_original_compare LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(STUDY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(JOINT_BUILD "${STUDY_ROOT}/build_direct_candidates" CACHE PATH "Existing joint build directory")
set(VKFFT_INCLUDE_ROOT "${STUDY_ROOT}/VkFFT/vkFFT" CACHE PATH "Original VkFFT include root")
set(JOINT_SOURCE "${JOINT_BUILD}/generated/direct_benchmark.cpp")
if(NOT EXISTS "${JOINT_SOURCE}" OR NOT EXISTS "${JOINT_BUILD}/generated/include/vkfft_trace_source_metadata.hpp")
  message(FATAL_ERROR "Existing joint generated source and metadata are required")
endif()
if(NOT EXISTS "${VKFFT_INCLUDE_ROOT}/vkFFT.h")
  message(FATAL_ERROR "Original VkFFT include root not found")
endif()
find_package(CUDAToolkit REQUIRED)
add_executable(vkfft_original_control "${JOINT_SOURCE}")
# Search the untouched VkFFT tree first. The generated include supplies only
# benchmark helper metadata and direct_candidate.hpp; its VkFFT copy is shadowed.
target_include_directories(vkfft_original_control PRIVATE "${VKFFT_INCLUDE_ROOT}" "${JOINT_BUILD}/generated/include" "${STUDY_ROOT}/test/plan_selection_study/instrumentation")
target_compile_definitions(vkfft_original_control PRIVATE VKFFT_BACKEND=1 VKFFT_STUDY_UNINSTRUMENTED=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
target_link_libraries(vkfft_original_control PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
if(NOT MSVC)
  target_compile_options(vkfft_original_control PRIVATE -O3)
endif()
