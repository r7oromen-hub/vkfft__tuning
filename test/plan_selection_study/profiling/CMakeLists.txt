cmake_minimum_required(VERSION 3.18)
project(vkfft_profile_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(STUDY_INSTRUMENTED_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../build_instrumented/generated/include" CACHE PATH "Existing accepted isolated instrumented include tree")
find_package(CUDAToolkit REQUIRED)
if(NOT EXISTS "${STUDY_INSTRUMENTED_INCLUDE}/vkFFT.h")
  message(FATAL_ERROR "Build/accept the independent instrumented headers first, then specify STUDY_INSTRUMENTED_INCLUDE")
endif()
add_executable(vkfft_profile_benchmark vkfft_profile_benchmark.cpp)
target_include_directories(vkfft_profile_benchmark PRIVATE "${STUDY_INSTRUMENTED_INCLUDE}" "${CMAKE_CURRENT_SOURCE_DIR}/../instrumentation")
target_compile_definitions(vkfft_profile_benchmark PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
target_link_libraries(vkfft_profile_benchmark PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
if(NOT MSVC)
  target_compile_options(vkfft_profile_benchmark PRIVATE -O3)
endif()
