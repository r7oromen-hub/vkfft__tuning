cmake_minimum_required(VERSION 3.18)
project(vkfft_plan_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PLAN_TUNING_CPU_ONLY "Build independent CPU reference test without CUDA" OFF)
set(VKFFT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../VkFFT" CACHE PATH "VkFFT repository root")
enable_testing()
add_executable(test_reference test_reference.cpp)
add_test(NAME cpu_reference COMMAND test_reference)
find_path(FFTW3_INCLUDE_DIR fftw3.h)
find_library(FFTW3_LIBRARY NAMES fftw3)
if(FFTW3_INCLUDE_DIR AND FFTW3_LIBRARY)
  add_executable(test_reference_fftw test_reference_fftw.cpp)
  target_include_directories(test_reference_fftw PRIVATE "${FFTW3_INCLUDE_DIR}")
  target_link_libraries(test_reference_fftw PRIVATE "${FFTW3_LIBRARY}")
  add_test(NAME cpu_reference_fftw_large COMMAND test_reference_fftw)
endif()
if(NOT PLAN_TUNING_CPU_ONLY)
  find_package(CUDAToolkit REQUIRED)
  add_executable(vkfft_plan_benchmark vkfft_plan_benchmark.cpp)
  target_include_directories(vkfft_plan_benchmark PRIVATE "${VKFFT_ROOT}/vkFFT")
  target_compile_definitions(vkfft_plan_benchmark PRIVATE VKFFT_BACKEND=1 CUDA_TOOLKIT_ROOT_DIR="${CUDAToolkit_LIBRARY_ROOT}")
  target_link_libraries(vkfft_plan_benchmark PRIVATE CUDA::cudart CUDA::cuda_driver CUDA::nvrtc)
  if(NOT MSVC)
    target_compile_options(vkfft_plan_benchmark PRIVATE -O3)
  endif()
endif()
